# 查询语句中支持表名.tbname

## 1. 背景

查询语句中目前对 select {table_name}.tbname 的支持不完善，有报错或者数据为空的情况，需完善。
[TD-29092](https://jira.taosdata.com:18080/browse/TD-29092)

## 4. 行为说明

### 4.1 基本说明

1. 支持普通表、系统表、超级表、子表的相关查询
2. 支持.tbname 在select, where, oreder by , group by , parition by 中出现
3. 支持 join 查询中的 {table_name}.tbname, 这时因为有多个表， {table_name} 的限定是必须的，不然报错.

### 4.2 场景举例

```json
超级表：
CREATE STABLE `st` (`ts` TIMESTAMP, `v1` INT) TAGS (`t1` INT);
CREATE STABLE `st2` (`ts` TIMESTAMP, `v1` INT) TAGS (`t1` INT);

子表：
CREATE TABLE `t1` USING `st` (`t1`) TAGS (1);
CREATE TABLE `t2` USING `st` (`t1`) TAGS (2);
CREATE TABLE `t21` USING `st2` (`t1`) TAGS (21);

普通表：
CREATE TABLE `nt` (`ts` TIMESTAMP, `v1` INT);

```

#### 4.2.1 普通表查询

1. 支持
select tbname, count(*) from nt;
select nt.tbname, count(*) from nt;
select tbname, count(*) from nt group by tbname;
select nt.tbname, count(*) from nt group by tbname;
select nt.tbname, count(*) from nt group by nt.tbname;

#### 4.2.2 系统表查询

1. 支持
select tbname, count(*) from information_schema.ins_dnodes;
select ins_dnodes.tbname, count(*) from information_schema.ins_dnodes;
select tbname, count(*) from information_schema.ins_dnodes group by tbname;
select ins_dnodes.tbname, count(*) from information_schema.ins_dnodes group by tbname;
select ins_dnodes.tbname, count(*) from information_schema.ins_dnodes group by ins_dnodes.tbname;

#### 4.2.3 子表查询

1. 支持：
select tbname, count(*) from t1;
select t1.tbname, count(*) from t1;
select tbname, count(*) from t1 group by tbname;
select t1.tbname, count(*) from t1 group by tbname;
select t1.tbname, count(*) from t1 group by t1.tbname;
1. 当表名不匹配时应当报错
select t1.tbname, count(*) from t2 group by t1.tbname;
select t1.tbname, count(*) from t1 group by t2.tbname;
select t2.tbname, count(*) from t1 group by t1.tbname;

#### 4.2.4 超级表查询

select tbname, count(*) from st group by tbname;
select tbname, count(*) from st partition by tbname;
select * from st where st.tbname="t1";
select tbname, ts from st where tbname="t2";
select tbname, ts from st where tbname="t2" order by tbname;
select tbname, ts from st where tbname="t2" order by st.tbname;
select tbname, count(*) from st where tbname="t2" group by tbname order by tbname;
select tbname, count(*) from st group by tbname order by tbname;
select tbname, count(*) from st group by st.tbname order by st.tbname;
// select st.tbname
select st.tbname, count(*) from st group by tbname;
select st.tbname, count(*) from st partition by tbname;
select st.tbname, ts from st where tbname="t2";
select st.tbname, ts from st where tbname="t2" order by tbname;
select st.tbname, ts from st where tbname="t2" order by st.tbname;
select st.tbname, count(*) from st where tbname="t2" group by tbname order by tbname;
select st.tbname, count(*) from st group by tbname order by tbname;
select st.tbname, count(*) from st group by st.tbname order by st.tbname;

#### 4.2.5 JOIN 查询

1. 子表之间
select t1.tbname, t2.tbname from t1, t2 where t1.ts=t2.ts and t1.tbname!=t2.tbname;
select t1.tbname, t2.tbname from t1, t2 where t1.ts=t2.ts and t1.tbname!=t2.tbname order by t1.tbname;
1. 超级表之间
select st.tbname, st2.tbname from st, st2 where st.ts=st2.ts and st.tbname!=st2.tbname order by st.tbname;
1. 子表与超级表之间
select t1.tbname, st2.tbname from t1, st2 where t1.ts=st2.ts and t1.tbname!=st2.tbname order by t1.tbname;
1. 普通表与超级表
select nt.ts, st.tbname from nt, st where nt.ts=st.ts  order by st.tbname
select nt.ts, t1.tbname from nt, t1 where nt.ts=t1.ts order by t1.tbname
