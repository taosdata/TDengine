#### 3. 测试语句

```sql
drop database if exists test;
create database test vgroups 1;
use test;
create stable st1(ts timestamp, i int) tags(id int, loc varchar(20));
create table st1s1 using st1 tags(1, 'beijing');
create table st1s2 using st1 tags(2, 'shanghai');
insert into st1s1 values(now, 1) st1s2 values(now, 2);
create stable st2(ts timestamp, i int) tags(id int, loc varchar(20));
create table st2s1 using st2 tags(1, 'beijing');
create table st2s2 using st2 tags(2, 'shanghai');
insert into st2s1 values(now, 1) st2s2 values(now, 2);

create user wxy pass '123456';

## 1. 切换wxy登录，测试“数据库无授权-表无授权”场景

## 2. 读写都无权限

select * from test.st1;
insert into test.st1s1 values(now, 10) test.st1s2 values(now, 20);

## 3. 读写都无权限

select * from test.st2; 
insert into test.st2s1 values(now, 10) test.st2s2 values(now, 20);

## 4. 切换root登录

grant read on test.st1 to wxy;

## 5. 切换wxy登录，测试“数据库无授权-表读授权”场景

## 6. 可以读st1全部子表数据，不能写

select * from test.st1; 
insert into test.st1s1 values(now, 10) test.st1s2 values(now, 20);

## 7. 对st2读写都无权限

select * from test.st2; 
insert into test.st2s1 values(now, 10) test.st2s2 values(now, 20);

## 8. 切换root登录

revoke read on test.st1 from wxy;
grant read on test.st1 with id = 1 to wxy;

## 9. 切换wxy登录，测试“数据库无授权-表读授权有标签条件”场景

## 10. 可以读st1中id=1的子表的数据，不能写

select tbname, * from test.st1; 
insert into test.st1s1 values(now, 10);
insert into test.st1s2 values(now, 20);

## 11. 对st2读写都无权限

select * from test.st2; 
insert into test.st2s1 values(now, 10) test.st2s2 values(now, 20);

## 12. 切换root登录

## 17. 切换wxy登录，测试“数据库无授权-表写授权有标签条件”场景

## 18. 可以写st1中id=1的子表的数据，不能读

## 19. 对st2读写都无权限

## 20. 切换root登录
```
